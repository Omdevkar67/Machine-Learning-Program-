# Step 1: Import Required Libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import DBSCAN
from sklearn.decomposition import PCA
from sklearn.metrics import silhouette_score


# Step 2: Load Seeds Dataset from downloaded file
file_path = "seeds_dataset.csv"

columns = [
    "Area",
    "Perimeter",
    "Compactness",
    "Kernel_Length",
    "Kernel_Width",
    "Asymmetry",
    "Groove_Length",
    "Class"
]

# The UCI seeds dataset is whitespace-delimited. Using modern sep=r'\s+' parameter.
df = pd.read_csv(file_path, sep=r"\s+", header=None)

# Assign standard column names
if df.shape[1] != 8:
    raise ValueError("Dataset must contain 7 features and 1 Class column.")

df.columns = columns

print("Dataset loaded successfully!")
print("Dataset Shape:", df.shape)

print("\nFirst Five Records:")
print(df.head())


# Step 3: Understand the Dataset
print("\nDataset Information:")
df.info()

print("\nStatistical Summary:")
print(df.describe())


# Step 4: Check Missing Values
print("\nMissing Values:")
print(df.isnull().sum())

# Convert columns to numeric and remove missing rows
df = df.apply(pd.to_numeric, errors="coerce")
df = df.dropna().reset_index(drop=True)

print("\nDataset Shape after cleaning:", df.shape)


# Step 5: Select Features
# Class column is not used for DBSCAN
X = df.drop("Class", axis=1)

print("\nFeatures used for DBSCAN:")
print(X.columns.tolist())


# Step 6: Standardize the Features
scaler = StandardScaler()

X_scaled = scaler.fit_transform(X)

print("\nData standardized successfully.")

print("\nFirst Five Standardized Records:")
print(pd.DataFrame(X_scaled, columns=X.columns).head())


# Step 7: Apply DBSCAN
dbscan = DBSCAN(
    eps=0.8,
    min_samples=5
)

labels = dbscan.fit_predict(X_scaled)

# Add cluster labels to dataset
df["Cluster"] = labels

print("\nDBSCAN Clustering Completed!")
print("Unique Labels:", np.unique(labels))


# Step 8: Identify Clusters and Noise Points
cluster_counts = df["Cluster"].value_counts().sort_index()

noise_count = int(np.sum(labels == -1))

n_clusters = len(set(labels)) - (1 if -1 in labels else 0)

print("\nNumber of Points in Each Cluster:")
print(cluster_counts)

print("\nNumber of Clusters:", n_clusters)
print("Number of Noise Points:", noise_count)


# Step 9: Display Noise Points
noise_points = df[df["Cluster"] == -1]

print("\nNoise Points:")
print(noise_points)


# Step 10: Reduce Dimensions using PCA
pca = PCA(n_components=2)

X_pca = pca.fit_transform(X_scaled)

print("\nExplained Variance Ratio:")
print(pca.explained_variance_ratio_)

print(
    "Total Variance Represented:",
    round(pca.explained_variance_ratio_.sum(), 4)
)


# Step 11: Visualize DBSCAN Clusters
plt.figure(figsize=(10, 7))

unique_labels = sorted(set(labels))

for label in unique_labels:

    cluster_points = X_pca[labels == label]

    if label == -1:
        marker = "x"
        size = 80
        label_name = "Noise"
    else:
        marker = "o"
        size = 60
        label_name = f"Cluster {label}"

    plt.scatter(
        cluster_points[:, 0],
        cluster_points[:, 1],
        marker=marker,
        s=size,
        label=label_name
    )

plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.title("DBSCAN Clustering of Seeds Dataset")
plt.legend()
plt.grid(True)
plt.show()


# Step 12: Calculate Silhouette Score
non_noise = labels != -1

if n_clusters >= 2 and np.sum(non_noise) > n_clusters:

    score = silhouette_score(
        X_scaled[non_noise],
        labels[non_noise]
    )

    print("\nSilhouette Score:", round(score, 4))

else:
    print("\nSilhouette Score cannot be calculated.")
    print("DBSCAN produced fewer than two valid clusters.")


# Step 13: Compare Actual Classes with DBSCAN Clusters
comparison = pd.crosstab(
    df["Class"],
    df["Cluster"]
)

print("\nActual Class vs DBSCAN Cluster:")
print(comparison)


# Step 14: Final DBSCAN Summary
print("\n" + "=" * 50)
print("DBSCAN CLUSTERING SUMMARY")
print("=" * 50)

print("Total Observations:", len(df))
print("Number of Features:", X.shape[1])
print("Number of Clusters:", n_clusters)
print("Number of Noise Points:", noise_count)

print("\nCluster-wise Distribution:")
print(cluster_counts)

print("=" * 50)


# Step 15: Experiment with Different Epsilon Values
eps_values = [0.5, 0.6, 0.7, 0.8, 0.9, 1.0]

results = []

for eps in eps_values:

    model = DBSCAN(
        eps=eps,
        min_samples=5
    )

    temp_labels = model.fit_predict(X_scaled)

    temp_clusters = (
        len(set(temp_labels))
        - (1 if -1 in temp_labels else 0)
    )

    temp_noise = int(np.sum(temp_labels == -1))

    results.append({
        "eps": eps,
        "clusters": temp_clusters,
        "noise_points": temp_noise
    })

results_df = pd.DataFrame(results)

print("\nDBSCAN Parameter Comparison:")
print(results_df)


# Final Conclusion
print("\nConclusion:")
print("DBSCAN successfully identified clusters and noise points")
print("in the Seeds Dataset using density-based clustering.")
