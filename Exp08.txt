
import os
os.environ["OMP_NUM_THREADS"] = "1"

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score

# Step 1: Load Dataset
data = pd.read_csv("Mall_Customers.csv")

# Remove spaces from column names
data.columns = data.columns.str.strip()

print("First 5 Records:")
print(data.head())

print("\nColumn Names:")
print(data.columns.tolist())

# Step 2: Select Features
X = data[['Annual Income (k$)', 'Spending Score (1-100)']].dropna().values

print("\nSelected Features:")
print(X[:5])

# Step 3: Elbow Method
wcss = []

for k in range(1, 11):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    model.fit(X)
    wcss.append(model.inertia_)

# Step 4: Plot Elbow Graph
plt.figure(figsize=(7, 5))
plt.plot(range(1, 11), wcss, marker='o')
plt.title("Elbow Method")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("WCSS")
plt.grid(True)
plt.show()

# Step 5: Silhouette Score
print("\nSilhouette Scores:")

silhouette_scores = []

for k in range(2, 11):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    labels = model.fit_predict(X)

    score = silhouette_score(X, labels)
    silhouette_scores.append(score)

    print("K =", k, "Score =", round(score, 3))

# Step 6: Plot Silhouette Scores
plt.figure(figsize=(7, 5))
plt.plot(range(2, 11), silhouette_scores, marker='o')
plt.title("Silhouette Score")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("Silhouette Score")
plt.grid(True)
plt.show()

# Step 7: Final K-Means Model
kmeans = KMeans(n_clusters=5, random_state=42, n_init=10)

clusters = kmeans.fit_predict(X)

# Step 8: Add Cluster Labels
data = data.dropna(subset=[
    'Annual Income (k$)', 'Spending Score (1-100)'
]).copy()

data['Cluster'] = clusters

print("\nClustered Dataset:")
print(data.head(10))

# Step 9: Number of Customers in Each Cluster
print("\nNumber of Customers in Each Cluster:")
print(data['Cluster'].value_counts().sort_index())

# Step 10: Visualize Clusters
plt.figure(figsize=(8, 6))

plt.scatter(
    X[:, 0],
    X[:, 1],
    c=clusters,
    s=50
)

plt.scatter(
    kmeans.cluster_centers_[:, 0],
    kmeans.cluster_centers_[:, 1],
    marker='X',
    s=200,
    label='Centroids'
)

plt.title("K-Means Clustering - Mall Customers")
plt.xlabel("Annual Income (k$)")
plt.ylabel("Spending Score (1-100)")
plt.legend()
plt.grid(True)
plt.show()
